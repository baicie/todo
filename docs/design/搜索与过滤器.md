## 2. 搜索模式

### 2.1 本地搜索（local 模式）

数据在 IndexedDB 中，搜索在内存中进行：

```typescript
class LocalSearchService {
  private tasks: Task[] = [];

  async loadTasks() {
    this.tasks = await storage.tasks.getTasks();
  }

  search(query: string): Task[] {
    if (!query.trim()) return [];

    const lower = query.toLowerCase();

    return this.tasks.filter(task => {
      return (
        task.title.toLowerCase().includes(lower) ||
        task.description?.toLowerCase().includes(lower) ||
        task.steps.some(step => step.title.toLowerCase().includes(lower)) ||
        task.category?.toLowerCase().includes(lower)
      );
    });
  }
}
```

### 2.3 搜索索引优化

当任务数量超过 1000 条时，纯内存搜索可能变慢。可引入 **Fuse.js** 进行模糊匹配：

```typescript
import Fuse from 'fuse.js';

const fuse = new Fuse(tasks, {
  keys: [
    { name: 'title', weight: 0.5 },
    { name: 'description', weight: 0.2 },
    { name: 'steps.title', weight: 0.2 },
    { name: 'category', weight: 0.1 },
  ],
  threshold: 0.3,        // 模糊程度
  includeScore: true,     // 返回匹配分数
  includeMatches: true,   // 返回匹配位置（用于高亮）
});

const results = fuse.search(query);
```

## 3. 搜索语法

### 3.1 快速筛选语法

支持类似 Linear/Notion 的井号筛选：

| 语法 | 含义 | 示例 |
|------|------|------|
| `#important` | 重要任务 | `项目 #important` |
| `#today` | 今天到期 | `报告 #today` |
| `#overdue` | 已过期 | `#overdue` |
| `#week` | 本周到期的 | `周会 #week` |
| `#blue` / `#red` | 按分类筛选 | `任务 #blue` |
| `in:[清单名]` | 在指定清单中 | `文档 in:工作` |
| `due:[日期]` | 指定到期日 | `报告 due:2026-04-25` |
| `is:completed` | 已完成任务 | `is:completed` |
| `is:incomplete` | 未完成任务 | `is:incomplete` |
| `has:steps` | 有步骤的任务 | `项目 has:steps` |
| `has:files` | 有附件的任务 | `has:files` |

### 3.2 解析器实现

```typescript
function parseSearchQuery(input: string): { query: string; filters: SearchFilter[] } {
  const filters: SearchFilter[] = [];
  let query = input;

  const filterPatterns = [
    { regex: /#important/gi, filter: { field: 'isImportant', value: true } },
    { regex: /#today/gi, filter: { field: 'dueDate', value: 'today' } },
    { regex: /#overdue/gi, filter: { field: 'dueDate', value: 'overdue' } },
    { regex: /#week/gi, filter: { field: 'dueDate', value: 'week' } },
    { regex: /#(blue|red|green|orange)/gi, filter: { field: 'category', value: null } },
    { regex: /in:(\S+)/gi, filter: { field: 'listId', value: null } },
    { regex: /is:completed/gi, filter: { field: 'isCompleted', value: true } },
    { regex: /is:incomplete/gi, filter: { field: 'isCompleted', value: false } },
    { regex: /has:steps/gi, filter: { field: 'hasSteps', value: true } },
    { regex: /has:files/gi, filter: { field: 'hasFiles', value: true } },
  ];

  for (const { regex, filter } of filterPatterns) {
    const match = input.match(regex);
    if (match) {
      filters.push(filter);
      query = query.replace(regex, '').trim();
    }
  }

  return { query, filters };
}
```

function highlightMatch(text: string, query: string): React.ReactNode {
  if (!query) return text;

  const parts = text.split(new RegExp(`(${escapeRegExp(query)})`, 'gi'));

  return parts.map((part, i) =>
    part.toLowerCase() === query.toLowerCase()
      ? <mark key={i} className="bg-yellow-200">{part}</mark>
      : part
  );
}

## 6. 搜索历史

```typescript
interface SearchHistory {
  query: string;
  timestamp: string;
  resultCount: number;
}

// 存储在 localStorage，限制最近 20 条
const MAX_HISTORY = 20;

function addSearchHistory(query: string, resultCount: number) {
  const history = getSearchHistory();

  // 去重：将相同查询移到最前
  const filtered = history.filter(h => h.query !== query);

  filtered.unshift({
    query,
    timestamp: new Date().toISOString(),
    resultCount,
  });

  // 限制数量
  if (filtered.length > MAX_HISTORY) {
    filtered.length = MAX_HISTORY;
  }

  localStorage.setItem('search-history', JSON.stringify(filtered));
}

// 显示最近搜索
function SearchHistoryDropdown({ onSelect }: Props) {
  const history = getSearchHistory();

  if (history.length === 0) return null;

  return (
    <div className="search-history">
      <div className="history-header">最近搜索</div>
      {history.map((h, i) => (
        <div key={i} className="history-item" onClick={() => onSelect(h.query)}>
          <ClockIcon /> {h.query}
          <span className="result-count">{h.resultCount} 个结果</span>
        </div>
      ))}
    </div>
  );
}
```

## 8. 搜索性能

| 优化手段 | 说明 |
|---------|------|
| 防抖 | 输入后 300ms 无变化再触发搜索，避免频繁查询 |
| 虚拟滚动 | 搜索结果超过 50 条时启用虚拟滚动渲染 |
| 截断展示 | 结果列表最多显示 100 条，底部显示"还有 X 条" |
| 内存缓存 | 本地搜索结果缓存在内存，切换页面后复用 |
| Web Worker | 大量任务（>5000）在 Web Worker 中执行搜索，避免 UI 卡顿 |
